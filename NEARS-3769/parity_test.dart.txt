import 'dart:typed_data';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:nears_dls/nears_dls.dart';

Future<Uint8List> shot(WidgetTester t, Widget w, TextDirection dir, double scale) async {
  final key = GlobalKey();
  await t.pumpWidget(MaterialApp(
    theme: ThemeData.light(),
    home: MediaQuery(
      data: MediaQueryData(size: const Size(400, 200), textScaler: TextScaler.linear(scale)),
      child: Directionality(textDirection: dir, child: Scaffold(
        body: Center(child: RepaintBoundary(key: key, child: SizedBox(width: 360, child: w))))),
    ),
  ));
  await t.pumpAndSettle();
  final ro = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  late Uint8List bytes;
  await t.runAsync(() async {
    final ui.Image img = await ro.toImage(pixelRatio: 2);
    bytes = (await img.toByteData(format: ui.ImageByteFormat.rawRgba))!.buffer.asUint8List();
  });
  return bytes;
}

void main() {
  final cases = <String, List<Widget Function()>>{
    'store_item_search': [
      () => NInput(controller: TextEditingController(), variant: NVariant.primary, hint: 'Search item in store', leadingIcon: 'search', keyboardType: TextInputType.text, textInputAction: TextInputAction.search, onSubmitted: (_) {}),
      () => NSearchField(controller: TextEditingController(), hint: 'Search item in store', leadingIcon: 'search', keyboardType: TextInputType.text, textInputAction: TextInputAction.search, onSubmitted: (_) {}),
    ],
    'location_dialog': [
      () => NInput(controller: TextEditingController(), textInputAction: TextInputAction.search, textCapitalization: TextCapitalization.words, keyboardType: TextInputType.streetAddress, hint: 'Search location', variant: NVariant.primary),
      () => NSearchField(controller: TextEditingController(), textInputAction: TextInputAction.search, textCapitalization: TextCapitalization.words, keyboardType: TextInputType.streetAddress, hint: 'Search location', leadingIcon: null),
    ],
  };
  // positive control: a genuinely different render must NOT compare equal
  testWidgets('control: pill vs field differs', (t) async {
    final a = await shot(t, NSearchField(controller: TextEditingController(), hint: 'x'), TextDirection.ltr, 1.0);
    final b = await shot(t, NSearchField(controller: TextEditingController(), hint: 'x', shape: NSearchShape.pill), TextDirection.ltr, 1.0);
    expect(listEquals(a, b), isFalse);
    // ignore: avoid_print
    print('PARITY control pill-vs-field differs bytes=${a.length}');
  });
  for (final e in cases.entries) {
    for (final dir in TextDirection.values) {
      for (final s in [1.0, 1.3]) {
        testWidgets('${e.key} ${dir.name} x$s', (t) async {
          final a = await shot(t, e.value[0](), dir, s);
          final b = await shot(t, e.value[1](), dir, s);
          int diff = 0;
          for (var i = 0; i < a.length && i < b.length; i++) { if (a[i] != b[i]) diff++; }
          // ignore: avoid_print
          print('PARITY ${e.key} ${dir.name} x$s lenA=${a.length} lenB=${b.length} diffBytes=$diff');
          expect(a.length, b.length);
          expect(diff, 0);
          expect(NSearchField.heightFor(TextScaler.linear(s)), NInput.heightFor(TextScaler.linear(s)));
        });
      }
    }
  }
}
