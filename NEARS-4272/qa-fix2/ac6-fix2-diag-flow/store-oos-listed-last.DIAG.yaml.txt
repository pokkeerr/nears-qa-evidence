# stock 0 (same SELECT); live label "SOLD OUT\n<name>\n<price>" (nav-guide §7a), so the name is
# matched anywhere in it.
- scrollUntilVisible:
    element: "[\\s\\S]*NEARS-1177 Beverage \\d\\d[\\s\\S]*"
    speed: 20
    visibilityPercentage: 60
    centerElement: true
    timeout: 90000
- assertVisible: "[\\s\\S]*NEARS-1177 Beverage \\d\\d[\\s\\S]*"
# Load every page first. items/latest pages are 13 items (31 items = pages 1-3), page 1 holds 3
# available + the 10 out-of-stock cards, and the app re-sorts only the pages it has loaded, so an
# order check on page 1 alone passes whatever pages 2-3 do (QA 25739). A page is requested when
# the grid reaches its end; with out-of-stock last, Beverage 10 is the last card at every stage.
# The small swipe back gives the next pass a real scroll delta (nav-guide §7a: a swipe already at
# max extent fires no load). 5 passes for 2 page loads leaves room for a slow response.
- repeat:
    times: 5
    commands:
      - scrollUntilVisible:
          element: "[\\s\\S]*NEARS-1177 Beverage 10[\\s\\S]*"
          speed: 40
          visibilityPercentage: 100
          timeout: 90000
      - waitForAnimationToEnd:
          timeout: 10000
      - swipe:
          start: 50%, 40%
          end: 50%, 70%
          duration: 600
# Back above the grid, by search (not by swipes, which could reach the top and pull-to-refresh
# page 1 again). The Recommended carousel sits right above All Products.
- scrollUntilVisible:
    element: "[\\s\\S]*Recommended For You[\\s\\S]*"
    direction: UP
    speed: 40
    visibilityPercentage: 60
    timeout: 90000
