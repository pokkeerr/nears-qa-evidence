### Helpers::parseZoneIds (fail-closed decoder):
    public static function parseZoneIds(string|null $raw): array
    {
        if (empty($raw) || $raw === 'null') {
            return [];
        }

        $decoded = json_decode($raw, true);

        if (!is_array($decoded) || !array_is_list($decoded)) {
            return [];
        }

        // array_is_list only checks indexing; reject object/bool/null/float elements before intval-mapping.
        foreach ($decoded as $v) {
            if (!is_int($v) && !(is_string($v) && ctype_digit($v))) {
                return [];
            }
        }

        return array_map('intval', $decoded);
    }
