#!/usr/bin/env python3
"""QA-only response-injecting reverse proxy (NEARS-3852 [8], extended from NEARS-3851 c1: + mutate_py).

listen :LISTEN -> forward to 127.0.0.1:UPSTREAM. Rules in RULES (JSON list), re-read
per request; a rule's `times` budget resets whenever the rules file changes.
Actions:
  replace          -> do NOT forward; answer {status, body}
  mutate_validate  -> forward, then set stores[store_id].pass=false/reason (+store_name), valid=false
  delay            -> forward after `delay` seconds (also honoured by the others)
Log line per request (no bodies/PII): method path status rule payment_method store_ids.
"""
import json, os, sys, time, threading, http.client
from http.server import ThreadingHTTPServer, BaseHTTPRequestHandler

LISTEN, UPSTREAM, RULES, LOG = int(sys.argv[1]), int(sys.argv[2]), sys.argv[3], sys.argv[4]
HOP = {'connection', 'keep-alive', 'proxy-authenticate', 'proxy-authorization', 'te', 'trailers',
       'transfer-encoding', 'upgrade', 'content-length', 'host', 'accept-encoding'}
_lock = threading.Lock()
_state = {'mtime': None, 'used': {}}


def load_rules():
    try:
        m = os.path.getmtime(RULES)
        with open(RULES) as f:
            rules = json.load(f)
    except Exception:
        return []
    with _lock:
        if m != _state['mtime']:
            _state['mtime'], _state['used'] = m, {}
    return rules


def take(rule):
    with _lock:
        used = _state['used'].get(rule['id'], 0)
        if rule.get('times', -1) >= 0 and used >= rule['times']:
            return False
        _state['used'][rule['id']] = used + 1
        return True


def log(line):
    with _lock, open(LOG, 'a') as f:
        f.write(time.strftime('%H:%M:%S ') + line + '\n')


class H(BaseHTTPRequestHandler):
    protocol_version = 'HTTP/1.1'

    def log_message(self, *a):
        pass

    def _go(self):
        n = int(self.headers.get('Content-Length') or 0)
        body = self.rfile.read(n) if n else b''
        path = self.path.split('?')[0]
        meta = ''
        if path.endswith('/group/validate') or path.endswith('/group/place'):
            try:
                j = json.loads(body or b'{}')
                meta = 'payment_method=%s store_ids=%s' % (
                    j.get('payment_method'), ','.join(str(s.get('store_id')) for s in j.get('stores', [])))
            except Exception:
                meta = 'payment_method=? (non-json)'
        if path.endswith('/get-Tax') or path.endswith('/delivery-quote') or path.endswith('/get-surge-price'):
            with _lock, open(LOG + '.bodies', 'a') as f:
                f.write(path + ' ' + body[:600].decode('utf-8', 'replace') + '\n')
        rule = None
        for r in load_rules():
            if r.get('enabled', True) and r.get('method', self.command) == self.command and r['path'] in path and (not r.get('body_contains') or r['body_contains'].encode() in body) and take(r):
                rule = r
                break
        if rule and rule.get('delay'):
            time.sleep(rule['delay'])
        if rule and rule['action'] == 'replace':
            out = json.dumps(rule['body']).encode()
            self.send_response(rule['status'])
            self.send_header('Content-Type', 'application/json')
            rid = self.headers.get('X-Request-Id')
            if rid:
                self.send_header('X-Request-Id', rid)
            self.send_header('Content-Length', str(len(out)))
            self.end_headers()
            self.wfile.write(out)
            log('%s %s -> %s INJECTED(replace:%s) %s' % (self.command, path, rule['status'], rule['id'], meta))
            return
        c = http.client.HTTPConnection('127.0.0.1', UPSTREAM, timeout=120)
        hdrs = {k: v for k, v in self.headers.items() if k.lower() not in HOP}
        hdrs['Host'] = '127.0.0.1:%d' % UPSTREAM
        c.request(self.command, self.path, body=body if body else None, headers=hdrs)
        resp = c.getresponse()
        data = resp.read()
        status = resp.status
        tag = ''
        if rule and rule['action'] == 'mutate_validate' and status == 200:
            try:
                j = json.loads(data)
                hit = False
                for s in j.get('stores', []):
                    if int(s.get('store_id')) == int(rule['store_id']):
                        s['pass'] = False
                        s['reason'] = rule['reason']
                        if 'store_name' in rule:
                            s['store_name'] = rule['store_name']
                        hit = True
                if hit:
                    j['valid'] = False
                    data = json.dumps(j).encode()
                    tag = ' INJECTED(mutate:%s store=%s reason=%s)' % (rule['id'], rule['store_id'], rule['reason'])
                else:
                    tag = ' MUTATE-MISS(%s store %s not in reply)' % (rule['id'], rule['store_id'])
            except Exception as e:
                tag = ' MUTATE-ERR(%s)' % type(e).__name__
        elif rule and rule['action'] == 'mutate_py':
            try:
                j = json.loads(data)
                env = {'j': j, 'status': status}
                exec(rule['py'], {}, env)
                data = json.dumps(env['j']).encode()
                status = env.get('status', status)
                tag = ' INJECTED(mutate_py:%s)' % rule['id']
            except Exception as e:
                tag = ' MUTATE-ERR(%s)' % type(e).__name__
        elif rule:
            tag = ' DELAYED(%s %ss)' % (rule['id'], rule.get('delay'))
        self.send_response(status)
        for k, v in resp.getheaders():
            if k.lower() not in HOP:
                self.send_header(k, v)
        self.send_header('Content-Length', str(len(data)))
        self.end_headers()
        self.wfile.write(data)
        extra = ''
        if path.endswith('/group/validate') or path.endswith('/group/place'):
            try:
                rj = json.loads(data)
                bj = json.loads(body or b'{}')
                extra = ' | req stores.dm_tips=%s partial=%s coupon=%s pm=%s | resp valid=%s cash_removed=%s amounts=%s stores=%s total_amount=%s' % (
                    [s.get('dm_tips') for s in bj.get('stores', [])], bj.get('partial_payment'), bj.get('coupon_code'), bj.get('payment_method'),
                    rj.get('valid'), rj.get('cash_removed'), json.dumps(rj.get('amounts')),
                    json.dumps([{k: s.get(k) for k in ('store_id', 'pass', 'reason', 'amounts')} for s in rj.get('stores', [])]) if isinstance(rj, dict) else '',
                    rj.get('total_amount') if isinstance(rj, dict) else '')
            except Exception as e:
                extra = ' | (unparsed %s)' % type(e).__name__
        log('%s %s -> %s%s %s%s' % (self.command, path, status, tag, meta, extra))

    do_GET = do_POST = do_PUT = do_DELETE = do_PATCH = _go


ThreadingHTTPServer(('0.0.0.0', LISTEN), H).serve_forever()
