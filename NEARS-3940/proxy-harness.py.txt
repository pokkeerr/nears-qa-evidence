LOG = os.path.join(HERE, 'proxy.log')
CTL = os.path.join(HERE, 'ctl.json')
lock = threading.Lock()

def log(line):

        with open(LOG, 'a') as f:
            f.write(line + '\n')

def ctl():

        return json.load(open(CTL))
    except Exception:
        return {}

    def log_message(self, *a): pass
